Database Connection (Before):
  conn = sqlite3.connect(DB_PATH)
  try:
      c = conn.cursor()
      # do stuff
      conn.commit()
  finally:
      conn.close()  ← Easy to forget!

Database Connection (After):
  with get_db() as conn:  ← Auto cleanup guaranteed
      c = conn.cursor()
      # do stuff
      conn.commit()


Decimal Parsing (Before):
  val = value.replace('.', '').replace(',', '.')
  amount = float(val)

Decimal Parsing (After):
  amount = GermanDecimalParser.parse(value)  ← Handles all formats


Regex Patterns (Before):
  if re.match(r"^\d{1,3}(\.\d{3})?,\d{2}$", value):
      # duplicated in multiple files

Regex Patterns (After):
  if InvoicePatterns.EURO_AMOUNT.value.match(value):  ← Central definition


Retry Logic (Before):
  for attempt in range(3):
      try:
          return api_call()
      except Exception:
          time.sleep(2 ** attempt)

Retry Logic (After):
  @retry_with_backoff(max_retries=3, initial_delay=1.0)
  def api_call():  ← Decorator handles retries
      pass
